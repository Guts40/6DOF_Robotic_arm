# ─── Send to Hardware ─────────────────────────────────────────────────────────
st.markdown("---")
col_send, col_stop, col_info = st.columns([1, 1, 2])


def build_command(joints):
    """Build serial command string for Arduino"""
    j = joints
    return f"W{j['waist']:03d}S{j['shoulder']:03d}E{j['elbow']:03d}P{j['wrist_pitch']:03d}R{j['wrist_roll']:03d}G{j['gripper']:03d}\n"


cmd = build_command(st.session_state.joints)

with col_send:
    if st.button("▶ SEND TO ARM", use_container_width=True):
        if st.session_state.arduino_connected:
            try:
                with serial.Serial(st.session_state.arduino_port, 9600, timeout=1) as ser:
                    ser.write(cmd.encode())
                st.success("✓ Command sent to Arduino")
            except Exception as e:
                st.error(f"Serial error: {e}")
        elif st.session_state.bt_connected:
            st.success("✓ Command queued for Bluetooth")
        else:
            st.info("Simulation: " + cmd.strip())

with col_stop:
    if st.button("⏹ EMERGENCY STOP", use_container_width=True):
        st.session_state.joints = {k: 90 for k in st.session_state.joints}
        st.session_state.joints["gripper"] = 0
        st.error("⚠ EMERGENCY STOP — All joints reset to home")
        st.rerun()

with col_info:
    st.markdown(f"""
    <div class="info-box">
        <span style="color:#4a6080;">SERIAL CMD:</span> <span style="color:#00ff88;">{cmd.strip()}</span>
    </div>
    """, unsafe_allow_html=True)

# ─── Arduino Firmware Reference ───────────────────────────────────────────────
with st.expander("📋 Arduino UNO Firmware Code (Upload this to your Arduino)"):
    st.code("""
#include <Servo.h>

Servo waist, shoulder, elbow, wristPitch, wristRoll, gripper;

void setup() {
    Serial.begin(9600);
    waist.attach(3);
    shoulder.attach(5);
    elbow.attach(6);
    wristPitch.attach(9);
    wristRoll.attach(10);
    gripper.attach(11);

    // Home position
    setAllServos(90, 90, 90, 90, 90, 0);
}

void loop() {
    if (Serial.available() > 0) {
        String cmd = Serial.readStringUntil('\\n');
        parseAndMove(cmd);
    }
}

void parseAndMove(String cmd) {
    // Format: W090S090E090P090R090G050
    int w  = cmd.substring(cmd.indexOf('W')+1, cmd.indexOf('S')).toInt();
    int s  = cmd.substring(cmd.indexOf('S')+1, cmd.indexOf('E')).toInt();
    int e  = cmd.substring(cmd.indexOf('E')+1, cmd.indexOf('P')).toInt();
    int p  = cmd.substring(cmd.indexOf('P')+1, cmd.indexOf('R')).toInt();
    int r  = cmd.substring(cmd.indexOf('R')+1, cmd.indexOf('G')).toInt();
    int g  = cmd.substring(cmd.indexOf('G')+1).toInt();

    // Map gripper 0-100 to servo angle 0-90
    setAllServos(w, s, e, p, r, map(g, 0, 100, 0, 90));

    // Send confirmation
    Serial.println("OK:" + cmd);
}

void setAllServos(int w, int s, int e, int p, int r, int g) {
    waist.write(w);
    shoulder.write(s);
    elbow.write(e);
    wristPitch.write(p);
    wristRoll.write(r);
    gripper.write(g);
}
""", language="cpp")
