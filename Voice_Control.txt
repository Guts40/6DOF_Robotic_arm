    <div id="history"></div>
</div>

<script>
const COMMANDS = {
    "go home":        "HOME",
    "close gripper":  "G100",
    "open gripper":   "G000",
    "shoulder up":    "S+15",
    "shoulder down":  "S-15",
    "elbow up":       "E+15",
    "elbow down":     "E-15",
    "rotate left":    "W-15",
    "rotate right":   "W+15",
    "stop":           "STOP",
    "wrist up":       "P+15",
    "wrist down":     "P-15",
    "start recording":"REC_START",
    "stop recording": "REC_STOP",
    "play sequence":  "PLAY",
};

let recognition, isListening = false;
const history = [];

function initSpeech() {
    const SpeechRec = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRec) {
        document.getElementById('status-line').textContent = "❌ Browser doesn't support Speech API (use Chrome)";
        return null;
    }
    const r = new SpeechRec();
    r.continuous = true;
    r.interimResults = true;
    r.lang = 'en-US';

    r.onstart = () => {
        document.getElementById('status-line').textContent = "🔴 LISTENING...";
        document.getElementById('mic-btn').classList.add('listening');
    };

    r.onresult = (e) => {
        let interim = '', final = '';
        for (let i = e.resultIndex; i < e.results.length; i++) {
            const t = e.results[i][0].transcript.toLowerCase().trim();
            if (e.results[i].isFinal) final = t;
            else interim = t;
        }
        document.getElementById('heard-text').textContent = final || interim || '—';
        if (final) processCommand(final);
    };

    r.onerror = (e) => {
        document.getElementById('status-line').textContent = "⚠ Error: " + e.error;
    };

    r.onend = () => {
        if (isListening) r.start(); // Keep going
    };
    return r;
}

function processCommand(text) {
    let matched = null;
    for (const [phrase, cmd] of Object.entries(COMMANDS)) {
        if (text.includes(phrase)) { matched = {phrase, cmd}; break; }
    }
    const resultEl = document.getElementById('cmd-result');
    if (matched) {
        resultEl.textContent = `✓ "${matched.phrase}" → ${matched.cmd}`;
        resultEl.style.color = '#00ff88';
        addHistory(text, matched.phrase, matched.cmd);
        // Post to Streamlit via URL params (for server-side action)
        const url = new URL(window.location);
        url.searchParams.set('voice_cmd', matched.cmd);
        window.history.replaceState({}, '', url);
    } else {
        resultEl.textContent = `⚠ Unknown: "${text}"`;
        resultEl.style.color = '#ffcc00';
    }
}

function addHistory(heard, phrase, cmd) {
    history.unshift({heard, phrase, cmd, time: new Date().toLocaleTimeString()});
    const el = document.getElementById('history');
    el.innerHTML = history.slice(0,10).map(h => 
        `<div class="hist-item"><span style="color:#4a6080;">[${h.time}]</span> 
         Heard: <span class="ht">"${h.heard}"</span> → 
         <span class="hc">${h.cmd}</span></div>`
    ).join('');
}

function toggleListening() {
    if (!recognition) recognition = initSpeech();
    if (!recognition) return;

    if (isListening) {
        recognition.stop();
        isListening = false;
        document.getElementById('status-line').textContent = "CLICK MICROPHONE TO START";
        document.getElementById('mic-btn').classList.remove('listening');
        document.getElementById('mic-btn').textContent = '🎤';
    } else {
        recognition.start();
        isListening = true;
        document.getElementById('mic-btn').textContent = '🔴';
    }
}
</script>
"""
st.components.v1.html(voice_html, height=460)

# ─── Manual Command Test ──────────────────────────────────────────────────────
st.markdown('<div class="section-header">◈ Test Commands Manually</div>', unsafe_allow_html=True)
col_t1, col_t2, col_t3 = st.columns(3)

quick_cmds = {
    "col1": [("🏠 Home", {"waist": 90, "shoulder": 90, "elbow": 90, "wrist_pitch": 90, "wrist_roll": 90, "gripper": 0}),
             ("🔒 Close Gripper", None),
             ("🔓 Open Gripper", None)],
    "col2": [("⬆ Shoulder Up", None), ("⬇ Shoulder Down", None), ("⬆ Elbow Up", None)],
    "col3": [("↩ Rotate Left", None), ("↪ Rotate Right", None), ("🔴 STOP", None)],
}

for col_el, col_key in zip([col_t1, col_t2, col_t3], ["col1", "col2", "col3"]):
    with col_el:
        for label, preset in quick_cmds[col_key]:
            if st.button(label, use_container_width=True, key=f"vc_{label}"):
                if preset:
                    st.session_state.joints.update(preset)
                else:
                    j = st.session_state.joints
                    if "Shoulder Up" in label:
                        j["shoulder"] = min(180, j["shoulder"] + 15)
                    elif "Shoulder Down" in label:
                        j["shoulder"] = max(0, j["shoulder"] - 15)
                    elif "Elbow Up" in label:
                        j["elbow"] = min(180, j["elbow"] + 15)
                    elif "Rotate Left" in label:
                        j["waist"] = max(0, j["waist"] - 15)
                    elif "Rotate Right" in label:
                        j["waist"] = min(180, j["waist"] + 15)
                    elif "Close" in label:
                        j["gripper"] = 100
                    elif "Open" in label:
                        j["gripper"] = 0
                    elif "STOP" in label:
                        j.update({k: 90 for k in j}); j["gripper"] = 0
                st.rerun()
