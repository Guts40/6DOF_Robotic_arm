# gesture_server.py — Run alongside Streamlit
# pip install opencv-python mediapipe pyserial

import cv2
import mediapipe as mp
import serial
import time

# Configuration
SERIAL_PORT = "COM3"      # Arduino port
CAM_INDEX   = 0           # 0 for USB webcam, or ESP32-CAM URL

mp_hands = mp.solutions.hands
mp_draw  = mp.solutions.drawing_utils

def count_fingers(hand_landmarks):
    tips = [8, 12, 16, 20]
    fingers = []
    # Thumb
    fingers.append(1 if hand_landmarks.landmark[4].x < hand_landmarks.landmark[3].x else 0)
    # Other fingers
    for tip in tips:
        fingers.append(1 if hand_landmarks.landmark[tip].y < hand_landmarks.landmark[tip-2].y else 0)
    return fingers

def classify_gesture(fingers):
    total = sum(fingers)
    if total == 0: return "FIST"
    if total == 5: return "OPEN"
    if fingers == [0,1,0,0,0]: return "INDEX_UP"
    if fingers == [0,1,1,0,0]: return "PEACE"
    if fingers == [1,0,0,0,1]: return "HANG_LOOSE"
    if fingers[0] and not any(fingers[1:]): return "THUMBS_UP"
    return f"FINGERS_{total}"

GESTURE_CMDS = {
    "FIST":       "G100",
    "OPEN":       "G000",
    "INDEX_UP":   "S+05",
    "PEACE":      "E+10",
    "HANG_LOOSE": "R+10",
    "THUMBS_UP":  "HOME",
}

cap = cv2.VideoCapture(CAM_INDEX)

try:
    ser = serial.Serial(SERIAL_PORT, 9600)
    print(f"Connected to Arduino on {SERIAL_PORT}")
except:
    ser = None
    print("No serial — simulation mode")

joints = {"waist":90, "shoulder":90, "elbow":90, "wrist_pitch":90, "wrist_roll":90, "gripper":0}

def apply_cmd(cmd, joints):
    if cmd == "HOME":
        return {k: 90 for k in joints} | {"gripper": 0}
    if cmd.startswith("G"): joints["gripper"] = int(cmd[1:])
    elif cmd.startswith("S+"): joints["shoulder"] = min(180, joints["shoulder"] + int(cmd[2:]))
    elif cmd.startswith("S-"): joints["shoulder"] = max(0,   joints["shoulder"] - int(cmd[2:]))
    elif cmd.startswith("E+"): joints["elbow"]    = min(180, joints["elbow"]    + int(cmd[2:]))
    elif cmd.startswith("R+"): joints["wrist_roll"]= min(180, joints["wrist_roll"]+ int(cmd[2:]))
    return joints

with mp_hands.Hands(max_num_hands=1, min_detection_confidence=0.7) as hands:
    last_gesture = None
    last_sent = 0

    while cap.isOpened():
        ret, frame = cap.read()
        if not ret: break

        frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
        result = hands.process(frame_rgb)

        gesture = "None"
        if result.multi_hand_landmarks:
            for lm in result.multi_hand_landmarks:
                mp_draw.draw_landmarks(frame, lm, mp_hands.HAND_CONNECTIONS)
                fingers = count_fingers(lm)
                gesture = classify_gesture(fingers)

        # Send command every 500ms if gesture changed
        now = time.time()
        if gesture != last_gesture and gesture in GESTURE_CMDS and (now - last_sent) > 0.5:
            cmd = GESTURE_CMDS[gesture]
            joints = apply_cmd(cmd, joints)
            serial_cmd = f"W{joints['waist']:03d}S{joints['shoulder']:03d}E{joints['elbow']:03d}"
            serial_cmd += f"P{joints['wrist_pitch']:03d}R{joints['wrist_roll']:03d}G{joints['gripper']:03d}"

            if ser:
                ser.write((serial_cmd + "\\n").encode())
            print(f"Gesture: {gesture} → CMD: {serial_cmd}")
            last_gesture = gesture
            last_sent = now

        # Display overlay
        cv2.putText(frame, f"Gesture: {gesture}", (10,30), cv2.FONT_HERSHEY_SIMPLEX, 1, (0,255,255), 2)
        cv2.imshow("Gesture Control — Press Q to quit", frame)

        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

cap.release()
cv2.destroyAllWindows()
if ser: ser.close()
""", language="python")
