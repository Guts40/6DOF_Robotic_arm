# ─── Arduino UNO ─────────────────────────────────────────────────────────────
st.markdown('<div class="section-header">◈ Arduino UNO (USB Serial)</div>', unsafe_allow_html=True)
col1, col2 = st.columns([2, 1])

with col1:
    # List available ports
    try:
        available_ports = [p.device for p in serial.tools.list_ports.comports()]
    except:
        available_ports = ["COM1", "COM3", "/dev/ttyUSB0", "/dev/ttyACM0"]

    if available_ports:
        port_options = available_ports
    else:
        port_options = ["No ports found"]

    arduino_port = st.selectbox("Serial Port", port_options,
                                index=0 if port_options else 0,
                                help="Select the COM port of your Arduino UNO")

    baud = st.selectbox("Baud Rate", [9600, 115200, 57600], index=0)

    col_a, col_b = st.columns(2)
    with col_a:
        if st.button("🔌 Connect Arduino", use_container_width=True):
            try:
                with serial.Serial(arduino_port, baud, timeout=1) as ser:
                    if ser.isOpen():
                        st.session_state.arduino_connected = True
                        st.session_state.arduino_port = arduino_port
                        st.success(f"✓ Connected to Arduino on {arduino_port}")
            except Exception as e:
                st.session_state.arduino_connected = True  # Simulation
                st.session_state.arduino_port = arduino_port
                st.warning(f"Simulation mode (No hardware): {e}")
    with col_b:
        if st.button("⏏ Disconnect", use_container_width=True, key="disc_arduino"):
            st.session_state.arduino_connected = False
            st.info("Arduino disconnected")

with col2:
    status = "🟢 CONNECTED" if st.session_state.arduino_connected else "🔴 DISCONNECTED"
    color = "#00ff88" if st.session_state.arduino_connected else "#ff4444"
    st.markdown(f"""
    <div class="conn-panel" style="text-align:center; padding: 30px 16px;">
        <div style="font-family:'Share Tech Mono',monospace; font-size:13px; color:{color}; letter-spacing:2px;">{status}</div>
        <div style="font-family:'Share Tech Mono',monospace; font-size:11px; color:#4a6080; margin-top:8px;">
            PORT: {st.session_state.arduino_port}<br>BAUD: 9600
        </div>
    </div>
    """, unsafe_allow_html=True)

#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

void setup() {
    Serial.begin(115200);
    SerialBT.begin("RoboArm_ESP32");  // Device name
    Serial.println("Bluetooth started - pairing name: RoboArm_ESP32");
}

void loop() {
    // Forward BT commands to Serial (connect to Arduino via UART)
    if (SerialBT.available()) {
        String cmd = SerialBT.readStringUntil('\\n');
        Serial.println("BT CMD: " + cmd);
        // Parse and forward to Arduino or control servos directly
        // Serial2.println(cmd);  // If forwarding to Arduino
    }

    // Forward Arduino responses to BT
    if (Serial.available()) {
        String resp = Serial.readStringUntil('\\n');
        SerialBT.println(resp);
    }
}
