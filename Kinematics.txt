# ─── FK Calculation ───────────────────────────────────────────────────────────
st.markdown('<div class="section-header">◈ Forward Kinematics — Joint Positions</div>', unsafe_allow_html=True)

joints = st.session_state.joints
w = math.radians(joints["waist"])
s = math.radians(joints["shoulder"] - 90)
e = math.radians(joints["elbow"] - 90)
wp = math.radians(joints["wrist_pitch"] - 90)
wr = math.radians(joints["wrist_roll"])


# Compute transformation matrices (simplified)
def dh_matrix(theta, alpha, a, d):
    """Denavit-Hartenberg transformation matrix"""
    ct, st_ = math.cos(theta), math.sin(theta)
    ca, sa = math.cos(alpha), math.sin(alpha)
    return np.array([
        [ct, -st_ * ca, st_ * sa, a * ct],
        [st_, ct * ca, -ct * sa, a * st_],
        [0, sa, ca, d],
        [0, 0, 0, 1]
    ])


T01 = dh_matrix(w, math.pi / 2, 0, 60)
T12 = dh_matrix(s, 0, L1, 0)
T23 = dh_matrix(e, 0, L2, 0)
T34 = dh_matrix(wp, math.pi / 2, L3, 0)
T45 = dh_matrix(wr, -math.pi / 2, 0, L4)

T02 = T01 @ T12
T03 = T02 @ T23
T04 = T03 @ T34
T05 = T04 @ T45

# Extract joint positions
p0 = np.array([0, 0, 0, 1])
p1 = T01 @ p0
p2 = T02 @ p0
p3 = T03 @ p0
p4 = T04 @ p0
p5 = T05 @ p0

positions = [np.array([0, 0, 0]), p1[:3], p2[:3], p3[:3], p4[:3], p5[:3]]
joint_names_fk = ["Base", "Waist", "Shoulder", "Elbow", "Wrist", "EE"]

    "Distance (mm)": f"{np.linalg.norm(pos):.2f}"
} for name, pos in zip(joint_names_fk, positions)]

    # Convert to rad/s
    omegas = np.array([math.radians(x) for x in [omega_waist, omega_shoulder, omega_elbow, omega_wp, omega_wr]])

with col_icr2:
    # Jacobian (simplified linear Jacobian for end effector velocity)
    # v_ee = J * omega
    # Using geometric approach

    ee = p5[:3]

    # For each joint, compute z-axis and position contribution
    z_axes = [
        np.array([0, 0, 1]),  # J1 waist: z-axis of frame 0
        T01[:3, :3] @ np.array([0, 0, 1]),  # J2
        T02[:3, :3] @ np.array([0, 0, 1]),  # J3
        T03[:3, :3] @ np.array([0, 0, 1]),  # J4
        T04[:3, :3] @ np.array([0, 0, 1]),  # J5
    ]
    origins = [np.array([0, 0, 0]), p1[:3], p2[:3], p3[:3], p4[:3]]

    # Linear velocity columns of Jacobian: z_i × (p_ee - p_i)
    J_linear = np.zeros((3, 5))
    for i in range(5):
        r = ee - origins[i]
        J_linear[:, i] = np.cross(z_axes[i], r)

    v_ee = J_linear @ omegas
    v_magnitude = np.linalg.norm(v_ee)

    # ICR for waist rotation (horizontal plane)
    if abs(omega_waist) > 0.01:
        icr_radius = v_magnitude / abs(math.radians(omega_waist)) if abs(omega_waist) > 0 else float('inf')
    else:
        icr_radius = float('inf')
