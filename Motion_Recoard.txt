# Save sequences next to app.py in a 'sequences' subfolder (works on Windows/Mac/Linux)
import sys
from pathlib import Path

SEQUENCES_DIR = str(Path(__file__).parent.parent / "sequences")
os.makedirs(SEQUENCES_DIR, exist_ok=True)


# Load saved sequences from disk
def load_sequences():
    seqs = []
    for f in os.listdir(SEQUENCES_DIR):
        if f.endswith(".json"):
            try:
                with open(os.path.join(SEQUENCES_DIR, f)) as fp:
                    seqs.append(json.load(fp))
            except:
                pass
    return sorted(seqs, key=lambda x: x.get("created", ""), reverse=True)


def save_sequence(name, frames):
    seq = {
        "name": name,
        "created": time.strftime("%Y-%m-%d %H:%M:%S"),
        "frame_count": len(frames),
        "duration_s": frames[-1]["t"] - frames[0]["t"] if len(frames) > 1 else 0,
        "frames": frames
    }
    path = os.path.join(SEQUENCES_DIR, f"{name.replace(' ', '_')}_{int(time.time())}.json")
    with open(path, "w") as f:
        json.dump(seq, f, indent=2)
    return seq

# ─── Replay Script ────────────────────────────────────────────────────────────
with st.expander("🐍 Sequence Replay Script (replay_sequence.py)"):
    st.code("""
# replay_sequence.py
# Usage: python replay_sequence.py my_sequence.json COM3

import json, serial, time, sys

seq_file = sys.argv[1] if len(sys.argv)>1 else "sequence.json"
port     = sys.argv[2] if len(sys.argv)>2 else "COM3"

with open(seq_file) as f:
    seq = json.load(f)

print(f"Replaying '{seq['name']}' — {seq['frame_count']} frames")

try:
    ser = serial.Serial(port, 9600)
except:
    ser = None
    print("No serial — simulation mode")

frames = seq['frames']
start = time.time()

for i, frame in enumerate(frames):
    # Timing
    target_t = frame['t']
    elapsed = time.time() - start
    if target_t > elapsed:
        time.sleep(target_t - elapsed)

    cmd  = f"W{frame['waist']:03d}"
    cmd += f"S{frame['shoulder']:03d}"
    cmd += f"E{frame['elbow']:03d}"
    cmd += f"P{frame['wrist_pitch']:03d}"
    cmd += f"R{frame['wrist_roll']:03d}"
    cmd += f"G{frame['gripper']:03d}\\n"

    print(f"  Frame {i+1}/{len(frames)}: {cmd.strip()}")
    if ser: ser.write(cmd.encode())

print("✓ Sequence complete")
if ser: ser.close()
""", language="python")
